$$
\begin{gathered}
\overline{\lim}a_n = \lim s_n, ~s_n := \sup_{m \geq n} a_m 
\end{gathered}
$$ 

Proof
$$
\begin{gathered}
\overline{\lim}a_n = \inf V := \{ v < a_n \text{ for a finitely many $n$'s} \} \\
\lim s_n = \inf V \\
\lim s_n \leq  \inf V \\
\text{Suppose } \inf < \lim s_n \\
\implies \exists~ v \in V \ni v < \lim s_n \\
\text{Since } v \in V, \implies v<a_n \text{ for a finitely many n's} \\
\implies \exists~ M > 0 \ni a_n \leq v~ \forall~  n \geq M \\
\implies s_n \leq v ~\forall~ n \geq  M \\
s_n = \sup \{ a_m  | n \geq m \} \text{ since } m \geq n \geq M \text{ then } a_m \leq v \\
\text{so } V \text{ is an upper bund of } \{ a_m, m \geq n \} \\
\text{So } s_n \leq v, \text{ for } n \geq M \\
\text{So } \lim s_n \leq v \unicode{x21af}
\end{gathered}
$$ 

Consider
$$
\overline{\lim} \{ x_n + y_n \} \leq \overline{\lim} x_n + \overline{\lim} y_n
$$ 

$$
\lim_{n \to \infty} \sup_{m \geq n} \{ x_m + y_m \} = \lim_{n \to \infty} \sup_{m \geq n} \{ x_m \} + \lim_{n \to \infty} \sup_{m \geq n} \{ y_m \}
$$ 

$$
\sup_{m \geq n} \{ x_m + y_m \} \leq 
\sup_{m \geq n} \{ x_n \} + 
\sup_{m \geq n} \{ y_n \}
\implies 
\lim_{n \to \infty} \sup_{m \geq n} \{ x_m + y_m \} = \lim_{n \to \infty} \sup_{m \geq n} \{ x_n \} + \lim_{n \to \infty} \sup_{m \geq n} \{ y_n \}
$$ 


$$ \sup_{m \geq n} \{ x_m \} + \sup_{m \geq n} \{ y_m \} \text{ is an upper bound of } \{ x_m + y_m, m \geq n \} $$ 

$$
\frac{x_m \leq \sup_{m \geq n} \{ x_m \}}{y_m \leq \sup_{m \geq n} \{ y_m \}}
$$
