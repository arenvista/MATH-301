		\begin{itemize}
			\item $(a>b) \land (b>c) \implies a>c$

\subsection*{Part (a): If $(a>0)$, then $(1/a>0)$ and $\frac{1}{1/a} = a$}
Suppose $a>0$ and $\frac{1}{a} \not > 0$. Consider the following:
\[
	\begin{aligned}
		\frac{1}{a}           & \not > 0         \\
		(\frac{1}{a}) \cdot a & \not > 0 \cdot a \\
		(\frac{1}{a}) \cdot a & \not > 0         \\
		1                     & \not > 0         \\
	\end{aligned}
\]
\begin{center}
	Clearly $1>0$ thus this is a contradiction. \\
	Thus we must accept the original proposition: $a>0 \implies \frac{1}{a}>0$.
\end{center}
Prove $\frac{1}{1/a}=a$:
\[
	\begin{aligned}
		\frac{1}{a} \cdot a                             & = 1         & \text{ Multiplicative Inverse }  \\ \\
		\frac{1}{a} \cdot \frac{1}{\frac{1}{a}}         & = 1         & \text{ Multiplicative Inverse }  \\ \\
		a \cdot \frac{1}{a} \cdot \frac{1}{\frac{1}{a}} & = 1 \cdot a & \text{ Multipy a on both sides } \\ \\
		1 \cdot \frac{1}{\frac{1}{a}}                   & = 1 \cdot a & \text{ Multiplicative Inverse  } \\ \\
		\frac{1}{\frac{1}{a}}                           & = a         & \text{ Multiplicative Identity } \\
	\end{aligned}
\]
\begin{center}
	Thus we can say $\frac{1}{1/a}=a$
\end{center}
Now we have proven if $(a>0)$, then $(1/a>0)$ and $\frac{1}{1/a} = a$.

\subsection*{Part a: If $0<c<1$, show that $0<c^2<c<1$.}
We can define $c < 1$ as $c = 1-\epsilon$, where $\epsilon = \frac{1}{n}$ for some $n \in \mathbb{N}$ such that $0 < \epsilon < 1$.

Observe the following:
\begin{align*}
	1         & > 1-\epsilon                                                     \\
	c         & = 1 - \frac{1}{n}                                                \\
	c^2       & = \left(1-\frac{1}{n}\right)^2 = 1 - \frac{2}{n} + \frac{1}{n^2}
	\intertext{We want to compare $c$ and $c^2$. Since $n \ge 1$ and $\epsilon > 0$, multiplying the inequality $c < 1$ by $c$ (where $c>0$) preserves the sign:}
	c \cdot c & < 1 \cdot c                                                      \\
	c^2       & < c
	\intertext{Thus, combining with our initial range:}
	0         & < c^2 < c < 1
\end{align*}

\subsection*{Part b: If $1<c$, show that $1<c<c^2$.}

We can define $c > 1$ as $c = 1+\epsilon$, where $\epsilon = \frac{1}{n}$ for some $n \in \mathbb{N}$.

Observe the following:
\begin{align*}
	1                & < 1+\epsilon                 \\
	c                & = 1 + \epsilon               \\
	c^2              & = (1+\epsilon)^2             \\
	                 & = 1 + 2\epsilon + \epsilon^2
	\intertext{Since $\epsilon > 0$, it must be that $\epsilon < 2\epsilon + \epsilon^2$. Therefore:}
	1 < 1 + \epsilon & < 1 + 2\epsilon + \epsilon^2 \\
	1 < c            & < c^2
\end{align*}
Since we defined $\epsilon > 0$, the statement $1 < c < c^2$ holds.

If $a>0$, $b>0$, and $n \in \mathbb{N}$, show that $a<b$ iff $a^n < b^n$ (induct.) \\ \\
$\forall ~a,b \in \mathbb{R}, a>0, b>0$ and $n \in \mathbb{N}$. \\
Consider $P(n): a^n < b^n \iff a < b$ \\
Observe $P(1)$:
\[
	\begin{aligned}
		a^1 < b^1 \implies a < b \\
		a < b \implies  a^1 < b^1
	\end{aligned}
\]
Suppose $P(k), k \in \mathbb{N}$:
\[
	a^k < b^k \implies  a < b
\]
Observe the following:
\[
	\begin{aligned}
		a^k         & < b^k                        \\
		a \cdot a^k & < b \cdot a^k & \text{ Eq 1} \\
		b \cdot a^k & < b \cdot b^k & \text{ Eq 2} \\
	\end{aligned}
\]
From Eq (1,2), we can derive the following
\[
	\begin{array}{cc}
		  & a^{k+1}  < b \cdot a^k               \\
		+ & b \cdot a^k  < b^{k+1}               \\
		\hline
		  & a^{k+1} + (ba^k)  < (ba^k) + b^{k+1}
	\end{array}
\]
Thus we can apply algera to derive the final statement.
\[
	\begin{aligned}
		a^{k+1} + (ba^k) + (-ba^k) & < (ba^k) + (-ba^k) + b^{k+1} \\
		a^{k+1}                    & <  b^{k+1}                   \\
	\end{aligned}
\]
\begin{center}
	Thus we conclude $a^k < b^k \implies a^{k+1} <  b^{k+1}$. Therefore $a<b$ iff $a^n<b^n$.
\end{center}
\section*{Question 7: Section 2.1-25}
\begin{center}
	Assuming the existence of roots, show that $c>1 \implies c^{1/m} < c^{1/n}$ iff $m>n$
\end{center}
\subsection*{Lemma 1: Integer Powers of $c > 1$}
First, we establish that for base $c > 1$, larger integer exponents yield larger values.
We show $c^b > c^a \iff b > a$ for $a,b \in \mathbb{N}$. \\ \\
\textbf{Forward ($b > a \implies c^b > c^a$):}
Let $b > a$. Then $b - a = k \in \mathbb{N}$ where $k \geq 1$.
Using Bernoulli's Inequality for $c = 1+x$ where $x > 0$:
\[ c^k = (1+x)^k \geq 1 + kx > 1 \]
Multiplying both sides by $c^a$ (which is positive):
\[ c^k \cdot c^a > 1 \cdot c^a \implies c^{a+k} > c^a \implies c^b > c^a \] \\
\textbf{Reverse ($c^b > c^a \implies b > a$):}
We proceed by contradiction. Assume $c^b > c^a$ but $b \leq a$.
\[
	\begin{aligned}
		\text{If } b = a & \implies c^b = c^a \quad (\text{Contradiction})        \\
		\text{If } b < a & \implies a - b = k \in \mathbb{N}, k \geq 1            \\
		                 & \implies c^a > c^b \quad (\text{By Forward proof})     \\
		                 & (\text{Contradiction}) \therefore c^b > c^a \iff b > a
	\end{aligned}
\]

\subsection*{Lemma 2: Roots of $c > 1$ exceed 1}
Let $c > 1$. Let $y = c^{1/k}$ for some $k \in \mathbb{N}$. We claim $y > 1$.
\begin{proof}
	Assume for contradiction that $y \leq 1$.
	Since $y > 0$, raising both sides to the power of $k$:
	\[ y \leq 1 \implies y^k \leq 1^k \implies c \leq 1 \]
	This contradicts the premise $c > 1$. Therefore, $c^{1/k} > 1$.
\end{proof}

\subsection*{Main Proof}
Let $N = mn$. Let $b = c^{1/N} = c^{1/mn}$.
From \textbf{Lemma 2}, we know $b > 1$.
We can rewrite the terms $c^{1/m}$ and $c^{1/n}$ using base $b$:
\[
	\begin{aligned}
		c^{1/m} & = (c^{\frac{1}{mn}})^n = b^n \\
		c^{1/n} & = (c^{\frac{1}{mn}})^m = b^m
	\end{aligned}
\]
We wish to show $c^{1/m} < c^{1/n} \iff m > n$.
Substituting the expressions in terms of $b$:
\[
	b^n < b^m \iff m > n
\]
Since $b > 1$, we apply \textbf{Lemma 1}:
\[
	b^m > b^n \iff m > n
\]
This is exactly the statement we needed to show.
