\begin{center}
	\textit{If $S := \{ \frac{1}{n}-\frac{1}{m} | n,m \in \mathbb{N} \}$ find $\inf S$ and $\sup S$}.
\end{center}
Consider for the following:
\[
	\begin{aligned}
		n \in \mathbb{N} & \implies 0 < \frac{1}{n} \leq 1   \\
		m \in \mathbb{N} & \implies -1 < -\frac{1}{m} \leq 1
	\end{aligned}
\]
\subsection*{Finding Supremum}
Find the maximum value of $\frac{1}{n}$ and minimum $\frac{1}{m}$.
\[
	\text{Considering the bounds:  }
	\left \{
	\begin{aligned}
		 & \implies \frac{1}{n} = 1  \\
		 & \implies -\frac{1}{m} = 0
	\end{aligned}
	\right .
\]
Note $\sup S = 1$ this statement holds if $1$ the smallest upperbound of $S$. \\

Consider: $\lim_{m \to \infty} (1-\frac{1}{m})$
\[ |(1-\frac{1}{m})-1| < \epsilon \iff |-\frac{1}{m} + (1-1) | < \epsilon \iff |-\frac{1}{m}| < \epsilon \iff |\frac{1}{m}| < \epsilon \iff \frac{|1|}{m} < \epsilon \\ \]
\[ \leftarrow m > \frac{1}{\epsilon}       \leftarrow \text{Archimedian Property}  \leftarrow \frac{1}{\epsilon} \in \mathbb{R} \]
\[ \therefore \lim_{m \to \infty} (1-\frac{1}{m})=-1 \]

\[ \leftarrow \text{Archimedian Property} \leftarrow \frac{1}{\epsilon} \in \mathbb{R} \]
\[ \therefore \lim_{n \to \infty} (\frac{1}{n}-1) = 1 \]

\[
	S = \{ s \in \mathbb{R} \mid s \geq 0 \text{ and } s^2 < 3 \}
\]
\subsection*{Existence of Supremum}
\[ 0^2 = 0 < 3 \implies 0 \in S \implies S \neq \emptyset \]
\[ \text{Let } \forall ~s \in S, s < 3 \text{ Suppose } \exists ~s \in S \ni s \geq 3 \implies  s^2 \geq  9 > 3 \text{ \lightning} \implies \text{S is bounded above.} \]
\[ S \neq \emptyset \land  S \text{ is bounded above } \implies \exists ~x \in \mathbb{R} \ni x=\sup S \text{ (By Completeness Property.)} \]

\subsection*{Trichotomy}
\[ \text{By Trichotomy, one and only one, of the following must be true:} \]
\[
	\begin{aligned}
		(x^2<3)  \quad\quad (x^2>3) \quad\quad (x^2=3)
	\end{aligned}
\]
\subsubsection*{(Case: $(x^2>3)$)}
\[
	\text{Suppose for the sake of contracdiction $x$ is the least upper bound of $S$}
\]
\[
	\text{Show: } \forall ~n \in \mathbb{N}, (x-\frac{1}{n})^2 > 3 \implies \forall ~s \in S, (x-\frac{1}{n})^2 > s^2  \implies  (x-\frac{1}{n}) > s
\]
Observe the following:
\[
	(x-\frac{1}{n})^2 = x^2 - \frac{2x}{n} + \frac{1}{n^2}
\]
\[
	x^2 - \frac{2x}{n} + \frac{1}{n^2} \xrightarrow{Dropping ~\frac{1}{n^2}} (x-\frac{1}{n})^2 > x^2- \frac{2x}{n}
\]
To hold supposition, the inequality must be stricly greater than three.
\[
	\begin{aligned}
		x^2 - \frac{2x}{n} & > 3            \\
		x^2 - 3            & > \frac{2x}{n} \\
		\frac{1}{x^2-3}    & < \frac{n}{2x} \\
		\frac{2x}{x^2-3}   & < n
	\end{aligned}
\]
Since $x^2 > 3 \implies  (x^2-3) > 0 \implies \exists ~n \ni x>\frac{1}{n} > 0$.

\[ (x-\frac{1}{2})^2 > 3 \implies \forall ~ s \in S, s^2 < 3 < (x-\frac{1}{n})^2         \]
\[ \implies x-\frac{1}{n} \text{ is an upper bound of } S \]
\[ x-\frac{1}{n} < x \implies x \text{ is not the least upper bound of $S$ \lightning} \]
\[ \therefore ~(x^2 \not > 3) \]
\subsubsection*{(Case: $(x^2<3)$)}
Observe the following:
\[
	(x+\frac{1}{n})^2 = x^2 + \frac{2x}{n} + \frac{1}{n^2}
\]
Note $\forall ~n \in \mathbb{N}, \frac{1}{n^2} \leq  \frac{1}{n}$
\[
	x^2 + \frac{2x}{n} + \frac{1}{n^2} \leq  x^2 + \frac{2x+1}{n}
\]
\[
	\begin{aligned}
		 & \implies x^2 + \frac{2x+1}{n} < 3         \\
		 & \implies \frac{2x+1}{n} < 3 - x^2         \\
		 & \implies \frac{n}{2x+1} > \frac{1}{3-x^2} \\
		 & \implies n > \frac{2x+1}{3-x^2}
	\end{aligned}
\]
\begin{center}
	As $x^2 < 3 \implies 3-x^2 > 0$. Therefore by Archimedian Property $\implies  (x+\frac{1}{n}) \in S$\\
	$\implies x$ is not an upperbound \lightning.
	\[ \therefore ~(x^2 \not < 3) \]
	\textbf{\textit{ As $x^2 \not < 3$ and $x^2 \not > 3$. By Trichotomy $x^2=3$ }}
\end{center}
