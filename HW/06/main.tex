\cbox{catgreen}{Cauchy Definition}{
	A sequence $X = (x_n)$ is called a \emph{Cauchy sequence} if the terms of the sequence get arbitrarily close to each other as the sequence progresses.
	More formally,
	\begin{align*}
		 & \text{For every } \epsilon > 0, \text{ there exists an integer } N_\epsilon > 0, \text{ such that} \\
		 & \qquad \forall\ m, n \geq N_\epsilon, \quad |x_n - x_m| < \epsilon.
	\end{align*}

	This means that after some index $N_\epsilon$, the distance between any two terms $x_n$ and $x_m$ (with $n, m$ large enough) is less than any chosen positive number $\epsilon$.
	Intuitively, the sequence becomes "clustered" together and the terms never drift far apart, no matter how far out in the sequence you go.
	\vspace{1em}

		Show directly from the definition that if $(x_n)$ and $(y_n)$ are Cauchy sequences, then $(x_n+y_n)$ and $(x_n y_n)$ are Cauchy sequences.
	}

		Let the sequence operations be defined as:
		\[
			x_n + y_n := \{ x_n+y_n \mid \forall~ n \} \quad \text{and} \quad x_n \cdot y_n := \{ x_n \cdot y_n \mid \forall~ n \}
		\]

		By the definition of Cauchy sequences:
		\begin{align*}
			|x_m - x_n| < \epsilon & \implies \forall~ \epsilon>0, \exists~ n_\epsilon \text{ s.t. } \forall~ m,n > n_\epsilon, |x_m - x_n| < \epsilon \\
			|y_m - y_n| < \epsilon & \implies \forall~ \epsilon>0, \exists~ n_\epsilon \text{ s.t. } \forall~ m,n > n_\epsilon, |y_m - y_n| < \epsilon
		\end{align*}

		For addition:
		\[
			|x_m+y_m - (x_n + y_n)| = |(x_m-x_n) + (y_m-y_n)| \leq |x_m - x_n| + |y_m - y_n| < 2\epsilon
		\]

		For multiplication, we establish bounds:
		\begin{align*}
			|x_m - x_n| & \leq |x_m| + |-x_n| \implies |y_m||x_m - x_n| \leq (|x_m| + |-x_n|)|y_m| \\
			|y_m - y_n| & \leq |y_m| + |-y_n| \implies |x_n||y_m - y_n| \leq (|y_m| + |-y_n|)|x_n|
		\end{align*}

		Now applying the triangle inequality trick for multiplication:
		\begin{align*}
			|x_m y_m - x_n y_n| & = |x_m y_m - \textcolor{red}{\mathbf{x_m y_n}} + \textcolor{red}{\mathbf{x_m y_n}} - x_n y_n| \\
			                    & = |x_m(y_m-y_n) + y_n(x_m-x_n)|                                                               \\
			                    & \leq |x_m||y_m-y_n| + |y_n||x_m-x_n|
		\end{align*}

		Because Cauchy sequences are bounded, we know:
		\begin{align*}
			|x_m - x_n| < \epsilon & \implies \exists~ M_x > 0 \text{ s.t. } \forall~ m, |x_m| < M_x \\
			|y_m - y_n| < \epsilon & \implies \exists~ M_y > 0 \text{ s.t. } \forall~ m, |y_m| < M_y
		\end{align*}

		Multiplying by these bounds:
		\begin{align*}
			M_x \cdot |y_m-y_n| & < \epsilon \cdot M_x \\
			M_y \cdot |x_m-x_n| & < \epsilon \cdot M_y
		\end{align*}

		Thus:
		\begin{align*}
			\implies |x_m y_m - x_n y_n|                                                    & < M_x |y_m-y_n| + M_y |x_m-x_n|                                                          \\

		\end{align*}

\cbox{catblue}{Section 3.5: 7}{
	Let $(x_n)$ be a Cauchy sequence such that $x_n$ is an integer for every $n \in \mathbb{N}$. Show that $(x_n)$ is ultimately constant.
}

Let $(x_n)$ be a Cauchy sequence where $x_n \in \mathbb{Z}$ for all $n \in \mathbb{N}$.

By the definition of a Cauchy sequence, choosing $\epsilon = 1$, there exists an $N \in \mathbb{N}$ such that for all $m, n \ge N$:
$$|x_m - x_n| < 1$$

Since $x_m, x_n \in \mathbb{Z}$, their absolute difference $|x_m - x_n|$ must be a non-negative integer. 

The only non-negative integer strictly less than $1$ is $0$. Therefore, for all $m, n \ge N$:
$$|x_m - x_n| = 0 \implies x_m = x_n$$

Thus, for all $n \ge N$, $x_n = x_N$. The sequence is ultimately constant.

\cbox{catblue}{Section 3.6: 7}{
Let $(x_n)$ and $(y_n)$ be sequences of positive numbers such that $\lim(x_n/y_n) = 0$.
\begin{itemize}
    \item (a) Show that if $\lim(x_n) = +\infty$, then $\lim(y_n) = +\infty$.
    \item (b) Show that if $(y_n)$ is bounded, then $\lim(x_n) = 0$.
\end{itemize}
}

\textbf{Infinite Limit:}
Given $\lim\left(\frac{x_n}{y_n}\right) = 0$ and $x_n, y_n > 0$.
Choose $\epsilon = 1$. $\exists N_1$ such that $n \ge N_1 \implies 0 < \frac{x_n}{y_n} < 1 \implies x_n < y_n$.

$\lim x_n = +\infty \implies \exists N_2 : n \ge N_2 \implies x_n > M$. Let $N = \max(N_1, N_2)$. Then $n \ge N \implies y_n > x_n > M$. So, $\lim y_n = +\infty$.

\textbf{Bounded $y_n$:}
Suppose $0 < y_n \le M\ \forall n$. For $\epsilon > 0$, $\exists N$ such that $n \ge N \implies 0 < \frac{x_n}{y_n} < \frac{\epsilon}{M}$, so $x_n < \frac{\epsilon}{M}y_n \le \epsilon$. Thus, $n \ge N \implies 0 < x_n < \epsilon \implies \lim x_n = 0$.


\begin{enumerate}
    \item[(a)] There exists $N_1$ such that if $n > N_1$, then $0 < x_n < y_n$. If $\lim(x_n) = \infty$ then $\lim(y_n) = \infty$.
    \item[(b)] Suppose that $|y_n| \leq M$ for some $M > 0$. Given $\varepsilon > 0$ there exists $N_{\varepsilon}$ such that if $n \geq N_{\varepsilon}$ then $0 < \frac{x_n}{y_n} \leq \frac{\varepsilon}{M}$. Therefore $|x_n| \leq (\varepsilon/M)y_n \leq \varepsilon$ for $n \geq N_{\varepsilon}$.
\end{enumerate}

\cbox{catpurple}{How to Pick Which to Use?}{
	% The sequential criterion vs epsilon-delta for limits
	\paragraph{Sequential criterion for limits}
	The sequential criterion is useful when it's easier to reason about the behavior of all sequences converging to a point, rather than working directly with $\varepsilon$ and $\delta$. Use the sequential criterion when:
	\begin{itemize}
		\item The function is complicated, or piecewise-defined, making $\varepsilon$-$\delta$ algebraic manipulations messy.
		\item You wish to prove a limit does \emph{not} exist (by producing sequences approaching the point with different limit values).
	\end{itemize}

	\paragraph{The $\varepsilon$-$\delta$ definition}
	The $\varepsilon$-$\delta$ definition is the standard and most direct way to prove limits exist. Use the $\varepsilon$-$\delta$ approach when:
	\begin{itemize}
		\item The function is well-behaved and algebra allows you to solve for $\delta$ in terms of $\varepsilon$.
		\item The problem asks explicitly for an $\varepsilon$-$\delta$ proof, or for instructive calculation of how $\delta$ depends on $\varepsilon$.
	\end{itemize}

}

\cbox{catgreen}{The Sequential Criterion for Limits}{
	\begin{theorem}[Equivalence of Epsilon-Delta and Sequential Limits]
		Let $A \subseteq \mathbb{R}$, let $f: A \to \mathbb{R}$ be a function, and let $c \in \mathbb{R}$ be a cluster point (or limit point) of $A$. Then the following two statements are logically equivalent:

		\begin{enumerate}
			\item \textbf{The Functional Limit:} $\displaystyle\lim_{x \to c} f(x) = L$
			\item \textbf{The Sequential Limit:} For every sequence $(x_n)$ in $A$ such that $x_n \neq c$ for all $n \in \mathbb{N}$ and $\lim_{n \to \infty} x_n = c$, the sequence of function values $(f(x_n))$ converges to $L$:

			      $\displaystyle \lim_{n \to \infty} f(x_n) = L$
		\end{enumerate}
	\end{theorem}}

Let $\epsilon > 0$ be given. Since $\epsilon > 0$, it follows that $\epsilon^2 > 0$. 

Because we are given that $\lim_{x \to c} (f(x))^2 = 0$, we know by definition that for any positive number (including $\epsilon^2$), there exists a $\delta > 0$ such that if $0 < |x - c| < \delta$, then:
$$|(f(x))^2 - 0| < \epsilon^2$$

This simplifies to:
$$|f(x)|^2 < \epsilon^2$$
Because both sides are non-negative, we can take the square root to get:
$$|f(x)| < \epsilon$$

Since we have shown that for any $\epsilon > 0$, there exists a $\delta > 0$ such that $0 < |x - c| < \delta \implies |f(x) - 0| < \epsilon$, we can conclude that $\lim_{x \to c} f(x) = 0$. \\

Consider the function
\[

\end{cases}
\]

Let $f(x) := g(x)$. Notice that $\lim_{x \to 0} (f(x))^2 = 1$, since $f(x)$ approaches either $1$ or $-1$ as $x$ approaches $0$ from the right or left, and $1^2 = (-1)^2 = 1$. However, $\lim_{x \to 0} f(x)$ does not exist, as $x \to 0^+$, $f(x) \to 1$, but as $x \to 0^-$, $f(x) \to -1$.
