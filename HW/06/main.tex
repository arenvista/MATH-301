\cbox{catgreen}{ Cauchy Definition }{
	A sequence $X = (x_n)$ is called a \emph{Cauchy sequence} if the terms of the sequence get arbitrarily close to each other as the sequence progresses. More formally,
	\[
		\begin{aligned}
			 & \text{For every } \epsilon > 0, \text{ there exists an integer } N_\epsilon > 0, \text{ such that} \\
			 & \qquad \forall\ m, n \geq N_\epsilon, \quad |x_n - x_m| < \epsilon.
		\end{aligned}
	\]

	This means that after some index $N_\epsilon$, the distance between any two terms $x_n$ and $x_m$ (with $n, m$ large enough) is less than any chosen positive number $\epsilon$. Intuitively, the sequence becomes "clustered" together and the terms never drift far apart, no matter how far out in the sequence you go. \vspace{1em}

	\begin{center} \textbf{\textit{ Example of a Cauchy Sequence }} \end{center}
	\begin{center}
		\begin{tikzpicture}[xscale=2, yscale=2]
			% Draw axis
			\draw[->] (0,0) -- (4.2,0) node[right] {$n$};
			\draw[->] (0,0.0) -- (0,2) node[above] {$x_n$};

			% Draw epsilon band
			\draw[dashed,red] (0,1.2) -- (4,1.2);
			\draw[dashed,red] (0,0.8) -- (4,0.8);
			\node[right,red] at (4,1.2) {$x^* + \epsilon$};
			\node[right,red] at (4,0.8) {$x^* - \epsilon$};

			% Limit line
			\draw[dotted] (0,1) -- (4,1);
			% \node[right] at (4,1) {$x^*$};

			\filldraw[blue] (\n,\y) circle (1.5pt);
			\node[below right, scale=0.85] at (\n,\y) {$x_{\lbl}$};

\cbox{catblue}{Section 3.5: 2b)}{
	Show directly from the definition that the following is a cauchy sequence:
	\[ x_n := \left( 1 + \frac{1}{2!} + \dots + \frac{1}{n!}\right) \] }

"English": $x_n$ is Cauchy if there's some index $N_\epsilon$ s.t. given any two indicies $m,k$ where $N_\epsilon \leq m,k$.
The differences of the two indicies is "sufficently small": $|x_m - x_k| < \epsilon$

Define terms:
\[
	x_m := \sum_{i=m ~|~ N_\epsilon < m}^{\infty} \frac{1}{m!}
	\quad \quad
	x_k := \sum_{i=k ~|~ N_\epsilon < k}^{\infty} \frac{1}{n!}
\]

Goal (Show the Following):
\[
	\forall~ \epsilon > 0,~
	\left|
	\sum_{i=m ~|~ N_\epsilon < m}^{\infty} \frac{1}{m!} -
	\sum_{i=k ~|~ N_\epsilon < k}^{\infty} \frac{1}{n!}
	\right| < \epsilon
\]

\[
	\equiv
	\left|
	1 + \frac{1}{2!} + \frac{1}{m!}
	-1 - \frac{1}{2!} - \frac{1}{n!}
	\right| < \epsilon
\]

Suppose $n<m$:
\[
	\equiv
	\left| \frac{1}{(n+1)!} + \dots  + \frac{1}{m!} \right| \leq
	\left| \frac{1}{2^{n+1}} + \frac{1}{2^{n+2}} + ~\dots ~ + \frac{1}{2^{m}} \right|
	\equiv \sum_{k=n+1}^{\infty} \left( \frac{1}{2} \right)^k
\]

\[
	\sum_{k=n+1}^{\infty} \left( \frac{1}{2} \right)^k = \frac{\frac{1}{2^n}}{1-\frac{1}{2}} = \frac{1}{2^{n-1}}
\]

\[
	-\frac{1}{n} \leq \frac{1}{2^{n-1}}  \leq \frac{1}{n} \implies \lim \frac{1}{2^{n-1}} = 0
\]

\[
	-\frac{1}{2^{n-1}} \leq |x_m - x_n| \leq \frac{1}{2^{n-1}}  \implies  \lim |x_m - x_n| = 0
\]

\[
	\lim |x_m - x_n| = 0 \implies \forall~ \epsilon > 0, \exists~  N_\epsilon > 0, ~|~  \forall\ m, n \geq N_\epsilon, |x_n - x_m| < \epsilon.
\]

\cbox{catblue}{Section 3.5: 3c)}{
	Show directly from the definition that the following is NOT a cauchy sequence:
	\[ ln(n) \] }

\cbox{catgreen}{Negation of Cauchy}{
	\[
		\exists~  \epsilon > 0 ~s.t.~ \forall~  n > 0, \exists~  m, k \geq n ~s.t.~  |x_n - x_m| \geq  \epsilon.
	\]
}

\[
	x_m := ln(2n) \quad \quad x_n := ln(n)
\]

Consider:
\[
	ln(2n) - ln(n) = ln(\frac{2n}{n}) = ln(2) \approx 0.6931
\]

\[
	\epsilon \in (0,1] \implies \exists~ \epsilon ~s.t.~ 0.6931 > \epsilon

\[
	\therefore x_n \text{ is not Cauchy}

\cbox{catblue}{Section 3.5: 4}{
	Show directly from the definition that if $(x_n)$ and $(y_n)$ are Cauchy sequences, then $(x_n+y_n)$ and $(x_n y_n)$ are Cauchy sequences }
\[
	x_n + y_n := \{ x_n+y_n ~\forall~ n \}
	\quad \quad
	x_n \cdot  y_n := \{ x_n \cdot  y_n ~\forall~ n \}
\]

\[
	\begin{gathered}
		|x_m - x_n| < \epsilon \implies \forall~ \epsilon>0, \exists~ n_\epsilon ~s.t.~ \forall~ m,n > n_\epsilon, |x_m - x_n| < \epsilon \\
		|y_m - y_n| < \epsilon \implies \forall~ \epsilon>0, \exists~ n_\epsilon ~s.t.~ \forall~ m,n > n_\epsilon, |y_m - y_n| < \epsilon
	\end{gathered}
\]

\[
	|x_m+y_m - x_n - y_n| = | (x_m-x_n) + (y_m-y_n) | \leq |x_m - x_n| + |y_m - y_n| < 2\epsilon
\]

\[
	\begin{gathered}
		|x_m - x_n | \leq  |x_m| + |-x_n| \implies |y_m|~|x_m - x_n| \leq  (|x_m| + |-x_n|)~|y_m| \\
		|y_m - y_n | \leq  |y_m| + |-y_n| \implies |x_n|~|y_m - y_n | \leq  (|y_m| + |-y_n|) ~|x_n| \\
	\end{gathered}
\]

\[
	\begin{gathered}
		| x_m y_m - x_n y_n | = |x_m y_m - {\color{red}{\mathbf{x_m y_n}}} + {\color{red}{\mathbf{x_m y_n}}} - x_n y_n | \\
		= |x_m ~ (y_m-y_n) + y_n ~(x_m-x_n)| \leq |x_m|\cdot |y_m-y_n| + |y_n| |x_m-x_n| \\
		|x_m|\cdot |y_m-y_n| + |y_n| |x_m-x_n|
	\end{gathered}
\]

\[
	\begin{gathered}
		|x_m - x_n| < \epsilon \implies \exists~ M_x ~s.t.~ \forall~ n, x_m < M_x \\
		|y_m - y_n| < \epsilon \implies \exists~ M_y ~s.t.~ \forall~ n, y_m < M_y 
	\end{gathered}
\]

\[
    \begin{gathered}
    M_x \cdot |x_m-x_n| < \epsilon \cdot  M_x \\
    M_y \cdot |y_m-y_n| < \epsilon \cdot  M_y \\
    \end{gathered}
\] 

\[
    \begin{gathered}
    \implies | x_m y_m - x_n y_n | < M_x \cdot |y_m-y_n| + M_y |x_m-x_n| < 2  M_y M_y \epsilon \\
    \frac{|y_m-y_n|}{M_y} + \frac{|x_m-x_n|}{M_x} < 2\epsilon \\
    \frac{|y_m-y_n|}{2M_y} + \frac{|x_m-x_n|}{2M_x} < \epsilon \\
    \end{gathered}
\]
