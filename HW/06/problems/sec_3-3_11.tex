\question{Section 3.3 : 11}{ Let $x_n := 1/1^2 + 1/2^2 + \cdots + 1/n^2$ for each $n \in \mathbb{N}$. Prove that $(x_n)$ is increasing and bounded, and hence converges. [*Hint*: Note that if $k \geq 2$, then $1/k^2 \leq 1/k(k - 1) = 1/(k - 1) - 1/k$.] }

Consider
\[
	\begin{gathered}
		\text{Let } b_n := \frac{1}{n^2} \implies  x_n := \sum_{n=1}^{\infty} b_n \\
		% \lim b_n = \lim_{n \to \infty}  \frac{1}{n^2} = 0  \\
		\text{Observe: } b_1 < b_1 + b_2 \implies  x_1 < x_2 \\
		\text{Suppose: } x_k < x_{k+1}  \implies \sum_{n=1}^{k} b_n< \sum_{n=1}^{k+1} b_n \\
		\implies b_{k+2} + \sum_{n=1}^{k} b_n < b_{k+2} + \sum_{n=1}^{k+1} b_n \implies x_{k+1} < x_{k+2}\\
		~\therefore~  \forall~ n \in  \mathbb{N},~ x_{n-1} < x_{n} \text{ Therefore we can say the sequence is increasing}
	\end{gathered}
\]

Note
\[
    \begin{gathered}
    S := \sum_{n=1}^{\infty} \frac{1}{n^2} = 1+ \sum_{n=2}^{\infty}  \frac{1}{n^2} \\
        \forall~ n \in  \mathbb{N}, 2 \leq n \implies  n(n-1) < n^2 \implies  \frac{1}{n^2} < \frac{1}{n(n-1)} 
    \end{gathered}
\] 

Consider
\[
    \sum_{n=1}^{\infty} \frac{1}{n^2}< 1+ \sum_{n=2}^{\infty} \left( \frac{1}{n-1} - \frac{1}{n} \right)
\] 

Observe (Canceling Terms)
\[
    \sum_{n=2}^{k} \left( \frac{1}{n-1} - \frac{1}{n} \right) = \left( 1 -  {\color{catorange} \frac{1}{2}}\right) + \left( {\color{catorange} \frac{1}{2}} - {\color{catgreen}\frac{1}{3}} \right) + \left( {\color{catgreen}\frac{1}{3}} - {\color{catorange}\frac{1}{4}} \right) \dots  + \left( \frac{1}{k-1} - \frac{1}{k} \right) = 1 - \frac{1}{k}
\] 
We could also take the gaussian summation of the individual terms confirmed with induction and show it equals to the same value. 

\[
    \lim_{k \to \infty}  1-\frac{1}{k} = 1 \implies  \sum_{n=1}^{\infty} \frac{1}{n^2} < 2
\] 

Therefore $x_n$ is mononic, increasing, and bounded above, therefore there must converge.
