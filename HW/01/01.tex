\section*{Question 1: From Blackboard}
\textbf{Proposition:} Let $S$ and $T$ be two sets. If both $S$ and $T$ are countable, then $S \cup T$ is countable.
\begin{proof} Suppose $S$ and $T$ are countable, and there exists $A,B \subseteq \mathbb{N}$, by definition there exists a bijective function:
	\[
		f: A \rightarrow S \text{ and } g: B \rightarrow (T-S)
	\]
	Observe we can construct $S \cup T$ as a disjoint set. Let $S, T$ be defined as follows:
	\[
		S = \{ s | s \in S \}, T = \{ t | t \in T \}
	\]
	Natrually then.
	\[
		S \cup T = \{ x | x \in S \lor x \in T \}
	\]
	Observe:
	\[
		S \cup T-S = \{ y | y \in S \lor (y \in T \land  y \not\in S) \}
	\]
	Consider the following statments (note reversible):
	\[
		\begin{aligned}
			 \iff & S \lor (y \in T \land  y \not\in S) \\
			 \iff & S \lor (y \in T \land  y \in S^c)   \\
			 \iff & y \in S \cup ( T \cap S^c)          \\
			 \iff & y \in (S \cup T) \cap (S \cup S^c)  \\
			 \iff & y \in (S \cup T) \cap U             \\
			 \iff & y \in S \cup T
		\end{aligned}
	\]
	\begin{center}
		Thus we can state that $S \cup T = S \cup T-S$. \\
	\end{center}
	To show $S \cup T$ is countable, we must construct a bijection $h: C \rightarrow S \cup (T-S), C \subseteq \mathbb{N}$ \\
	\textbf{Define}  $h: C \rightarrow S \cup (T-S)$:
	\[
		h(n) =
		\begin{cases}
			f\left(\frac{n}{2}\right)   & \text{if } n \text{ is even} \\
			g\left(\frac{n+1}{2}\right) & \text{if } n \text{ is odd}
		\end{cases}
	\]
    \begin{blockquote}
        \textbf{Aside: } $S,(T-S)T$ must be disjoint. \\
        \quad $\exists ~n_1, n_2 \in C, n_1 \neq n_2 \ni h(n_1) = f(n_1) = h(n_2) = g(n_2)$ 
    \end{blockquote}
	\begin{center}
		To prove the proposition we must show that $h$ is \textit{surjective} and \textit{injective}.
	\end{center}
	\subsection*{Surjectivity}
	\textbf{Goal:} We want to show that $\forall ~y \in S \cup T, \exists ~n \in C \ni h(n) = y$
	\subsubsection*{Case 1 ($y \in S$):}
	Let $y \in S$. Since $f$ is surjective, $\exists ~ k \in C$ s.t. $f(k) = y$. \\
	Let $n=2k$. By definition $n$ is even. Observe:
	\[
		h(n) = f(\frac{2k}{2}) = f(k) = y
	\]
	\subsubsection*{Case 2 ($y \in T$):}
	Let $y \in T$. Since $g$ is surjective, $\exists ~k \in C \ni g(k) = y$ \\
	Let $n=2k-1$. Then $n$ is odd. Observe:
	\[
		h(n) = g(\frac{(2k-1)+1}{2}) = g(\frac{2k}{2}) = g(k) = y
	\]
	\begin{blockquote}
		\textbf{Recall:} \textit{Countable Set Implications} \\
		The following statements are equivalent:
		\begin{center}
			$S$ is countable. $\iff \exists $ surjection of $\mathbb{N}$ onto $S$. $\iff \exists $ an injection of $S$ into $\mathbb{N}$.
		\end{center}
	\end{blockquote} \vspace{1em} \noindent
    \begin{center}
        As every element $y \in S \cup T ~\exists ~n \in C \ni h(n) = y \implies h: C \rightarrow S \cup (T-S)$ is surjective.
    \end{center}
	Note since $h(n): C \rightarrow S \cup (T-S)$ is surjective and $C \subseteq \mathbb{N}$. \\
	$\implies$ Thus we can validate a condition in "Countable Set Implications" ($\exists $ surjection $\mathbb{N}$ onto $S$). \\
	$\implies$ It means there must be an injection of $S \cup (T-S) \rightarrow \mathbb{N}$.
	\begin{center}
		This means that $S \cup (T-S)$ must be countable.
	\end{center}
\end{proof}

\section*{Question 2: Sec 1.3 (q1)}
\textbf{Question:} Prove that a nonempty set $T_1$ is finite iff $\exists$ a bijection from $T_1$ onto a finite set $T_2$.
\subsubsection*{Case 1 $\rightarrow$  :}
\textbf{Note:} To prove this by "Pigeon Hole Principle" we must suppose $|T_1| = |T_2|$ else there does not exist a way to construct a bijection.
\begin{proof} (\textbf{Show:} If $T_1$ is finite. Then $\exists$ a $f: T_1 \rightarrow  T_2$, where $T_2$ is finite and $f$ is bijective) \\
	Suppose $T_1$ is finite. Consider a function $f: T_1 \rightarrow T_2$ defined as:
	\[
		f := \{ (a_i,b_j) | a_i \in T_1 \text{ and } b_j \in T_2 \}
	\]
	Let $a_i$ be an arbitrary element defined to be (where all elments are distict):
	\[
		a_i = a_1, a_2, a_3, ..., a_{|T_1|},
	\]
	Let $b_i$ be an arbitrary element defined to be (where all elments are distict):
	\[
		b_i = b_1, b_2, b_3, ..., b_{|T_2|},
	\]
	Clearly:
	\[
		\forall ~b \in T_2, \exists ~a \in T_1 \ni f(a)= b
	\]
    \begin{center}
        \textit{Thus $f$ is surjective.}
    \end{center}
	Clearly:
	\[
		f(x) = f(y) \implies x = y
	\]
    \begin{center}
        \textit{Thus $f$ is injective.}
    \end{center}
	\textbf{Conclusion:} As $f$ is both injective and surjective, given $T_1$ is finite then clearly there exists a bijection from $T_1$ onto $T_2$.
\end{proof}
\subsubsection*{Case 2 $\leftarrow$ :}
\begin{proof} (\textbf{Show:} If $\exists$ a $f: T_1 \rightarrow  T_2$, where $T_2$ is finite and $f$ is bijective. Then $T_1$ is finite.) \\
	Suppose $\exists$ a $f: T_1 \rightarrow  T_2$, where $T_2$ is finite and $f$ is bijective. \\
	For the sake of contradiction supose $T_1$ is infinite. \\
	As $f$ is bijective we can contruct an arbitrary mapping as:
	\[
		f := \{ (a_i,b_j) | a_i \in T_1 \text{ and } b_j \in T_2 \}
	\]
	Let $a_i$ be an arbitrary element defined to be (where all elments are distict):
	\[
		a_i = a_1, a_2, a_3, ..., a_{|T_1|},
	\]
	Let $b_i$ be an arbitrary element defined to be (where all elments are distict):
	\[
		b_i = b_1, b_2, b_3, ..., b_{|T_2|},
	\]
	However as $T_1$ is infinite and $T_2$ is finite then clearly $|T_2| < |T_1|$. \\
	Thus for the function $f: T_1 \rightarrow T_2$ to be a function the following must be true (by "Pigeon Hole Principle"):
	\[
		f(x) = f(y) \implies x \neq y
	\]
	\textbf{Conclusion:} This is a contradiction, thus we can say: if $\exists$ a $f: T_1 \rightarrow  T_2$, where $T_2$ is finite and $f$ is bijective. Then $T_1$ is finite.
\end{proof}
\begin{center}
    As we have proved both Case 1 and Case 2 we have shown that a nonempty set $T_1$ is finite iff $\exists$ a bijection from $T_1$ onto a finite set $T_2$.
\end{center}

	\subsection*{Setup}
	We want to find a set of \textit{all odd integers greater than 13}. Let's define this as $S$ (where $i \in \mathbb{N}$):
	\[

	\begin{center} (\textit{Note: all elements defined are distict}) \end{center}

	\begin{align*}
		n_1 = 1 \\
		n_2 = 2 \\
		n_3 = 3 \\
		...     \\
		n_j = j
	\end{align*}
	\begin{center} (\textit{Note: all elements defined are distict}) \end{center}
	Lets construct a function that maps $\mathbb{N}$ to $S$:
	\[
		f: \mathbb{N} \rightarrow S = \{ (s_i,n_j) | s_i \in S \land n_j \in \mathbb{N}  \}
	\]
	Observe this is modeling the function as:
	\[
		f(n) = 13 + 2n

	% prove via php; i.e. compare cardinalites; => if |A| = |B| imp exi a func f: A -> B s.t. f is bijective; question asks to exhibit (thus must construct a func. )
	\subsection*{Surjectivity}
	\begin{proof} \textbf{Goal:} Show that $\forall ~s \in S, \exists ~n \in \mathbb{N} \ni f(n) = s$ \\
		Let $s \in S, n \in \mathbb{N}$:
		\[
			s_n = 13 + 2n = f(n)
		\]
		By algebra:
		\begin{align*}
			s_n & = 13 + 2n  \\
			2n  & = s_n - 13
		\end{align*}
		Recall $\forall ~s \in S$, $s$ must be an odd integer greater than 13. Thus there must be some $k \in \mathbb{N}$ s.t.
		\[
			s = 2k+1
		\]
		By subsitution:
		\begin{align*}
			2n & = s_n - 13     \\
			2n & = (2k+1) - 13  \\
			2n & = 2k +(1 - 13) \\
			2n & = 2k - 12      \\
			n  & = k - 6
		\end{align*}
		Now we must ask if $n$ as defined to be as $n=k-6$ escapes the set $\mathbb{N}$ as $n$ is defined to be $n \in \mathbb{N}$. \\
		As $s$ is defined to be \textit{greater than 13}. We know there exists a greatest lower bound 15 that satisfies the criterion. By subsitution:
		\begin{align*}
			s      & > 13   \\
			s      & = 2k+1 \\
			2k+1   & > 13   \\
			2k+1-1 & > 13-1 \\
			2k     & > 12   \\
			k      & > 6    \\
		\end{align*}
		As $k > 6$ then this means the $n = k-6$ must always be greater than zero. As all operations retain closure properties of integers, we can expect the value to be an interger. By def all integers greater than zero must belong to $\mathbb{N}$. Thus we can say $n \in \mathbb{N}$. Thus the properties of surjectivity hold.
		\textbf{Backsolve:}
		\begin{align*}
			f(k-6) & = 13 + 2(k-6) \\
			f(k-6) & = 13 + 2k-12  \\
			f(k-6) & = 13-12 +2k   \\
			f(k-6) & = 1 + 2k = s
		\end{align*}
		\begin{center}
			We have shown that $\forall ~s \in S, \exists ~n \in \mathbb{N} \ni f(n) = s$
		\end{center}
	\end{proof}
	\subsection*{Injectivity}
	\begin{proof} \textbf{Goal:} Show that given $a,b \in \mathbb{N}$ and $f(a) = f(b) \implies a = b$ \\
		Let $a,b \in \mathbb{N}$. Suppose:
		\[
			f(a) = f(b)
		\]
		By subsitution:
		\begin{align*}
			2a+1   & = 2b+1   \\
			2a+1-1 & = 2b+1-1 \\
			2a     & = 2b     \\
			a      & = b
		\end{align*}
		\begin{center}
			Thus we have shown that given $a,b \in \mathbb{N}$ and $f(a) = f(b) \implies a = b$
		\end{center}
	\end{proof}
	\subsection*{Conclusion}
	As $f$ is both injective and surjective we have shown there exists a bijection between $\mathbb{N}$ and the set of all odd integers greater than 13.
\end{proof}

\textbf{Question:} Exhibit a bijection between $\mathbb{N}$ and a proper subset of itself.\\
Let's construct a set $E$, a set of all even natural numbers, and let $k \in \mathbb{N}$:
\[
	E := \{ n | n=2k \}
\]
\begin{center} Clearly $1 \in \mathbb{N}$ and $1 \not\in E$ as $1$ is not even. Therefore $E \subset \mathbb{N}$. \end{center}
To prove a bijection exists, we must construct a function $f: \mathbb{N} \rightarrow E$ that is surjective and injective.
Let's construct the function as follows, let $n \in \mathbb{N}, v \in E$:
\[
	f(n) = 2n = v
\]
\subsection*{Surjectivity}
\begin{proof} \textbf{Goal:} Show that $\forall ~s \in S, \exists ~n \in \mathbb{N} \ni f(n) = s$ \\
	Let $v \in E, n \in \mathbb{N}$. We can say the following:
	\[
		v = 2n \implies \frac{v}{2} = n
	\]
	Recall that $v \in E$, which constricts the elements of $E$ to be even, as defined.
	Let $k \in \mathbb{N}$ and $v = 2k$. Thus we can rearrange the equation as follows:
	\[
		\frac{v}{2} = n \implies \frac{2k}{2} = n \implies k = n
	\]
	\begin{center} Now notice we have not violated any closure properties thus the solution remains valid. \end{center}
	\textbf{Backsolve:} to confirm validity.
	\[
		f(k) = 2k = v
	\]
	\begin{center} \textit{Recall: we had defined $v = 2k$, so this is the expected answer.}\\ \end{center}
	\textbf{Conclusion:}
	\begin{center}
		Thus, we have show that $\forall ~s \in S, \exists ~n \in \mathbb{N} \ni f(n) = s$ \\
		Therefore $f$ is surjective.
	\end{center}
\end{proof}
\subsection*{Injectivity}
\begin{proof} \textbf{Goal:} Show that given $a,b \in \mathbb{N}$ and $f(a) = f(b) \implies a = b$ \\
	Let $a,b \in \mathbb{N}$ and $f(a) = f(b) \implies a = b$. Observe:
	\begin{align*}
		f(a)         & = f(b)         \\
		2a           & = 2b           \\
		\frac{2a}{2} & = \frac{2b}{2} \\
		a            & = b
	\end{align*}
	\begin{center}
		Thus we have shown that given $a,b \in \mathbb{N}$ and $f(a) = f(b) \implies a = b$ \\
		Therefore $f$ is injective.
	\end{center}
\end{proof}
\subsection*{Conclusion}
As $f$ is both injective and surjective we have shown there exists a bijection between $\mathbb{N}$ and the set of all even natural numbers.

\section*{Question 5: Sec 1.3 (q7)}
\textbf{Question:} Prove that a set $T_1$ is denumerable iff there is a bijection from $T_1$ onto a denumerable set $T_2$
\subsubsection*{Case 1 $\rightarrow$  :}
\textbf{Prove:} Given a denumerable set $T_1$ we can construct a bijection onto a denumerable set $T_2$.
\begin{proof}
	Suppose $T_1$ is denumerable. This implies a bijective function $f$ exists:
	\[
		f: \mathbb{N} \rightarrow T_1
	\]
	Consider the existance of another denumerable set $T_2$. Likewise a bijective fuction $g$ must also exist:
	\[
		g: \mathbb{N} \rightarrow T_2
	\]
	As these functions are bijective, this implies the existance of an inverse function for each.
	\[
		f^{-1}: T_1 \rightarrow \mathbb{N} \text{ and } g^{-1}: T_2 \rightarrow \mathbb{N}
	\]
	\textbf{Goal:} we want to show there exists a function $h: T_1 \rightarrow T_2$ that is bijective. \\
	\textit{Approach: consider the change in space}
	\[
		T_1 \rightarrow \mathbb{N} \rightarrow T_2
	\]
	Consider the composition function:
	\[
		g \circ f^{-1}: T_1 \rightarrow T_2
	\]
	From definition above we know $Img(f^{-1}) = \mathbb{N}$, via the implication of bijection. \\
	We also know that $g: \mathbb{N} \rightarrow T_2$ is bijective. \\
	Thus we can say the following is a valid bijective function:
	\[
		g: Img(f^{-1}) \rightarrow T_2 \equiv g: \mathbb{N} \rightarrow T_2
	\]
	\textbf{Conclusion:} Thus we have shown given a denumerable set $T_1$ we can construct a bijection towards another denumerable set $T_2$. \\
\end{proof}
\subsubsection*{Case 2 $\leftarrow$ :}
\textbf{Prove:} Given a denumerable set $T_2$, if a bijection exists from $T_1$ onto $T_2$ then $T_1$ must be denumerable.
\begin{proof}
	Suppose $T_2$ is a denumerable set. Thus, there must exist a bijection from $T_2$ and $\mathbb{N}$. Thus there must exist some functions:
	\[
		f: \mathbb{N} \rightarrow T_2 \text{ and } f^{-1}: T_2 \rightarrow \mathbb{N}
	\]
	Suppose there exists a bijection from $T_1$ onto $T_2$. We can define the following functions:
	\[
		g: T_1 \rightarrow T_2 \text{ and } g^{-1}: T_2 \rightarrow T_1
	\]
	\textit{Approach: consider the change in space}
	\[
		T_1 \rightarrow T_2 \rightarrow \mathbb{N}
	\]
	We can construct the composition:
	\[
		f^{-1} \circ g: T_1 \rightarrow \mathbb{N}
	\]
	Consider the image set:
	\[
		Img(g) = T_2
	\]
	By subsitution:
	\[
		f^{-1}(Img(g)) = \mathbb{N}
	\]
	As these functions are bijective, we can consider their inverses.
	\[
		f(\mathbb{N}) = T_2 \implies g(T_2) = T_1 \implies g(f(\mathbb{N})) = T_1
	\]
	This means there exists a bijective function from $\mathbb{N} \rightarrow T_1$. Therefore, $T_1$ must be denumerable.\\
	\textbf{Conclusion:} We have shown given a denumerable set $T_2$, if a bijection exists from $T_1$ onto $T_2$ then $T_1$ must be denumerable.
\end{proof}
