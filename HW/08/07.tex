\textbf{Statement (1):} Observe $\forall~ n \in \mathbb{N}, x_n \in E \implies f(x_n) = g(x_n)$.

As this is true we are left to conclude $f(x_0) = g(x_0)$. Therefore $x_0 \in E$.

\section{Section 5.4 - 6}
Show that if $f$ and $g$ are uniformly continuous on $A \subseteq \mathbb{R}$ and both bounded on $A$, then their product $fg$ is uniformly continuous on $A$.

Since $f$ and $g$ are bounded on $A$, there exist constants $M_f, M_g \ge 0$ such that
\[
|f(x)| \le M_f \quad \text{and} \quad |g(x)| \le M_g \qquad \text{for all } x \in A.
\]

To prove that $fg$ is uniformly continuous on $A$, let $\varepsilon > 0$ be given. Since $f$ is uniformly continuous on $A$, there exists $\delta_1 > 0$ such that
\[
|x-y|<\delta_1 \implies |f(x)-f(y)| < \frac{\varepsilon}{2(M_g+1)}
\qquad (x,y\in A).
\]
Similarly, since $g$ is uniformly continuous on $A$, there exists $\delta_2 > 0$ such that
\[
|x-y|<\delta_2 \implies |g(x)-g(y)| < \frac{\varepsilon}{2(M_f+1)}
\qquad (x,y\in A).
\]

Let $\delta = \min\{\delta_1,\delta_2\}$. If $x,y\in A$ and $|x-y|<\delta$, then
\[
\begin{aligned}
|f(x)g(x)-f(y)g(y)|
&= |f(x)g(x)-f(x)g(y)+f(x)g(y)-f(y)g(y)| \\
&\le |f(x)|\,|g(x)-g(y)| + |g(y)|\,|f(x)-f(y)| \\
&\le M_f\,|g(x)-g(y)| + M_g\,|f(x)-f(y)| \\
&< M_f\cdot \frac{\varepsilon}{2(M_f+1)} + M_g\cdot \frac{\varepsilon}{2(M_g+1)} \\
&< \frac{\varepsilon}{2} + \frac{\varepsilon}{2} = \varepsilon.
\end{aligned}
\]

Therefore, $fg$ is uniformly continuous on $A$.

If $f$ is uniformly continuous on $A \subseteq \mathbb{R}$, and $|f(x)| \geq k > 0$ for all $x \in A$, show that $1/f$ is uniformly continuous on $A$.

\[
\begin{gathered}
\text{Let } x,y \in A \\
|f(x)| \geq k > 0 \implies 0 < \frac{1}{|f(x)|} \leq \frac{1}{k} \\
|f(y)| \geq k > 0 \implies 0 < \frac{1}{|f(y)|} \leq \frac{1}{k} \\
\end{gathered}
\] 

We can construct the following:
\[
\left| \frac{1}{f(x)} - \frac{1}{f(y)} \right| \leq \frac{1}{k^2} \left| f(x)-f(y) \right|
\] 

Let $\epsilon > 0$. Because $f$ is uniformly continuous on $A$, there exists a $\delta > 0$ such that for all $x, y \in A$ with $|x-y| < \delta$, we have $|f(x) - f(y)| < k^2 \epsilon$.
Therefore, $|x-y| < \delta \implies \left| \frac{1}{f(x)} - \frac{1}{f(y)} \right| \leq \frac{1}{k^2} (k^2 \epsilon) = \epsilon$. Thus, $1/f$ is uniformly continuous on $A$.
