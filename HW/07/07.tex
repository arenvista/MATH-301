\begin{cbox}{catred}{Theorem 4.2.4: Algebraic Limit Theorem}
	\begin{center}
		\textit{Let $A \subseteq \mathbb{R}$, let $f$ and $g$ be functions on $A$ to $\mathbb{R}$, let $c \in \mathbb{R}$ be a cluster point of $A$. Let $b \in \mathbb{R}$.}
	\end{center} \vspace{1em}
	\textbf{(a)} \textit{If $\lim_{x \to c} f = L$ and $\lim_{x \to c} g = M$, then:}
	\begin{align*}
		\lim_{x \to c} (f + g) & = L + M, & \lim_{x \to c} (f - g) & = L - M, \\
		\lim_{x \to c} (fg)    & = LM,    & \lim_{x \to c} (bf)    & = bL.
	\end{align*}
	\textbf{(b)} \textit{If $h : A \to \mathbb{R}$, if $h(x) \neq 0$ for all $x \in A$, and if $\lim_{x \to c} h = H \neq 0$, then}
	\[ \lim_{x \to c} \left( \frac{f}{h} \right) = \frac{L}{H}. \]
\end{cbox}
